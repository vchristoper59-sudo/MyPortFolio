import java.util.Scanner;
import java.util.Random;

public class GuessTheWord {

    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);
        Random random = new Random();

        String[] words = {
            "computer",
            "keyboard",
            "internet",
            "programming",
            "student",
            "java",
            "school",
            "monitor"
        };

        String playAgain = "yes";

        System.out.println("================================");
        System.out.println("       GUESS THE WORD GAME");
        System.out.println("================================");

        while (playAgain.equalsIgnoreCase("yes")) {

            String answer = words[random.nextInt(words.length)];
            int attempts = 5;
            boolean correct = false;

            System.out.println("\nGuess the secret word!");
            System.out.println("Hint: It starts with the letter '"
                    + answer.charAt(0) + "'");
            System.out.println("Number of letters: " + answer.length());
            System.out.println("You have 5 attempts.");

            while (attempts > 0 && !correct) {

                System.out.print("\nEnter your guess: ");
                String guess = input.nextLine().trim();

                if (guess.isEmpty()) {
                    System.out.println("Please enter a word.");
                    continue;
                }

                if (guess.equalsIgnoreCase(answer)) {

                    System.out.println("\nCORRECT!");
                    System.out.println("You guessed the word!");
                    correct = true;

                } else {

                    attempts--;

                    System.out.println("Wrong guess!");

                    if (attempts > 0) {
                        System.out.println("Attempts remaining: " + attempts);
                    } else {
                        System.out.println("Game Over!");
                        System.out.println("The correct word was: " + answer);
                    }
                }
            }

            System.out.print("\nDo you want to play again? (yes/no): ");
            playAgain = input.nextLine().trim();

        }

        System.out.println("\n================================");
        System.out.println(" THANK YOU FOR PLAYING!");
        System.out.println("================================");

        input.close();
    }
}
